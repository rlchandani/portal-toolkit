# Portal Toolkit v1.5 — Live Mirror & Control Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the ~1.5s PNG "peek" screen with a real-time, low-latency screen mirror of the Portal that can also be controlled (tap, swipe, long-press, hardware keys, text) from the phone — a focused "scrcpy-lite" — plus an honest photo-count guidance hint on the Screensaver tab.

**Architecture:** Two independent pipelines run over the single existing shared `Dadb` (ADB multiplexes logical streams — proven by `LogcatViewModel` streaming while the health monitor pings). A **video pipeline** opens a long-lived `screenrecord --output-format=h264 -` `AdbStream`, feeds raw H.264 bytes to a `MediaCodec` decoder rendering to a `Surface`. A **control pipeline** maps phone gestures/buttons to short-lived `input ...` `shell()` calls.

**Tech Stack:** Kotlin, Jetpack Compose (Material3), `dev.mobile:dadb` 1.2.10 (`AdbStream` + `okio.BufferedSource`), Android `MediaCodec` (async callback mode), `SurfaceView` via `AndroidView`.

```kotlin
package com.portalinstaller.data.tools

import com.portalinstaller.data.adb.shellQuote

/** Taste hardware ale Portal-ului si codurile lor KeyEvent. */
enum class PortalKey(val code: Int) {
    BACK(4),
    HOME(3),
    RECENTS(187),
    POWER(26),
    VOLUME_UP(24),
    VOLUME_DOWN(25)
}

/** Aruncata cand screenrecord nu porneste (suprafata securizata / DRM / neacceptat). */
class MirrorUnsupportedException(message: String) : Exception(message)

/**
 * Logica pura pentru controlul Portal-ului: mapeaza coordonate din spatiul
 * view in spatiul dispozitivului (tinand cont de benzile de letterbox de la
 * ContentScale.Fit) si construieste comenzile `input ...`. Fara ADB, fara
 * Android — complet testabila.
 */
object InputInjector {

    /**
     * Mapeaza un punct din view (px) in coordonate de dispozitiv (px). Suprafata
     * foloseste Fit, deci exista benzi negre pe o axa; cartografiem doar in
     * dreptunghiul de continut si taiem (clamp) la marginile dispozitivului.
     */
    fun mapPoint(
        viewX: Float,
        viewY: Float,
        viewW: Int,
        viewH: Int,
        devW: Int,
        devH: Int
    ): Pair<Int, Int> {
        if (viewW <= 0 || viewH <= 0 || devW <= 0 || devH <= 0) return 0 to 0
        // Scala Fit: cel mai mic factor care incape continutul in view.
        val scale = minOf(viewW.toFloat() / devW, viewH.toFloat() / devH)
        val contentW = devW * scale
        val contentH = devH * scale
        val barX = (viewW - contentW) / 2f
        val barY = (viewH - contentH) / 2f
        val devX = ((viewX - barX) / scale).toInt().coerceIn(0, devW)
        val devY = ((viewY - barY) / scale).toInt().coerceIn(0, devH)
        return devX to devY
    }

    fun tapCommand(devX: Int, devY: Int): String = "input tap $devX $devY"

    fun swipeCommand(x1: Int, y1: Int, x2: Int, y2: Int, durationMs: Int): String =
        "input swipe $x1 $y1 $x2 $y2 $durationMs"

    fun longPressCommand(devX: Int, devY: Int, durationMs: Int = 600): String =
        swipeCommand(devX, devY, devX, devY, durationMs)

    fun keyeventCommand(key: PortalKey): String = "input keyevent ${key.code}"

    /**
     * `input text` trateaza spatiul special: il inlocuim cu %s, apoi citam
     * intregul argument cu shellQuote ca sa fie sigur in shell.
     */
    fun textCommand(raw: String): String {
        val withSpaces = raw.replace(" ", "%s")
        return "input text ${shellQuote(withSpaces)}"
    }
}
```

```kotlin
package com.portalinstaller.data.tools

import dadb.Dadb
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.currentCoroutineContext
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.flow
import kotlinx.coroutines.flow.flowOn
import kotlinx.coroutines.isActive
import kotlinx.coroutines.withContext

/** Rezolutia dispozitivului asa cum o raporteaza `wm size`. */
data class MirrorSize(val width: Int, val height: Int)

/**
 * Deschide si detine stream-ul H.264 `screenrecord` de pe Portal si expune
 * octetii ca Flow. screenrecord are o limita de timp (o setam la 180s), deci
 * la sfarsit de stream redeschidem transparent in aceeasi bucla, ca sursa sa
 * para continua. Toata munca de I/O ruleaza pe Dispatchers.IO.
 */
object ScreenMirrorManager {

    private const val TIME_LIMIT_SECONDS = 180
    private const val CHUNK_SIZE = 64 * 1024
    private val SIZE_REGEX = Regex("""(\d+)x(\d+)""")

    /** Prefera "Override size" (rezolutia afisata), altfel "Physical size". */
    fun parseWmSize(output: String): MirrorSize? {
        val lines = output.lines().map { it.trim() }
        val override = lines.firstOrNull { it.startsWith("Override size:") }
        val physical = lines.firstOrNull { it.startsWith("Physical size:") }
        val chosen = override ?: physical ?: return null
        val m = SIZE_REGEX.find(chosen) ?: return null
        val w = m.groupValues[1].toIntOrNull() ?: return null
        val h = m.groupValues[2].toIntOrNull() ?: return null
        return MirrorSize(w, h)
    }

    suspend fun deviceSize(dadb: Dadb): MirrorSize = withContext(Dispatchers.IO) {
        val resp = dadb.shell("wm size")
        parseWmSize(resp.allOutput)
            ?: throw MirrorUnsupportedException("Nu pot citi rezolutia (wm size)")
    }

    /**
     * Flow rece: la colectare deschide screenrecord si emite bucati de octeti
     * pana la anulare. La sfarsitul normal (limita de timp) redeschide. Inchide
     * mereu AdbStream-ul in finally. Daca screenrecord esueaza imediat, arunca
     * MirrorUnsupportedException.
     */
    fun stream(dadb: Dadb, size: MirrorSize): Flow<ByteArray> = flow {
        val command = "shell:screenrecord --output-format=h264 " +
            "--time-limit=$TIME_LIMIT_SECONDS " +
            "--size ${size.width}x${size.height} -"
        while (currentCoroutineContext().isActive) {
            val adbStream = dadb.open(command)
            try {
                val source = adbStream.source
                val buffer = ByteArray(CHUNK_SIZE)
                var emittedAny = false
                while (currentCoroutineContext().isActive) {
                    val read = source.read(buffer)
                    if (read == -1) break
                    if (read > 0) {
                        emittedAny = true
                        emit(buffer.copyOf(read))
                    }
                }
                // Daca stream-ul s-a inchis fara niciun octet, screenrecord a
                // esuat (suprafata securizata / neacceptat) — semnalam.
                if (!emittedAny) {
                    throw MirrorUnsupportedException("screenrecord nu a produs cadre")
                }
            } finally {
                runCatching { adbStream.close() }
            }
        }
    }.flowOn(Dispatchers.IO)
}
```

```kotlin
package com.portalinstaller.data.tools

import android.media.MediaCodec
import android.media.MediaFormat
import android.view.Surface
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import java.util.concurrent.ConcurrentLinkedQueue

/**
 * Decodor H.264 (video/avc) in mod asincron pe baza de callback. Primeste bucati
 * de octeti prin submit(), le pune intr-o coada si le trimite in buffer-ele de
 * intrare pe masura ce codec-ul le ofera (onInputBufferAvailable). Cadrele
 * decodate sunt eliberate direct pe Surface. Nu stie nimic despre ADB.
 */
class H264Decoder {

    private val _firstFrameRendered = MutableStateFlow(false)
    val firstFrameRendered: StateFlow<Boolean> = _firstFrameRendered.asStateFlow()

    private val _error = MutableStateFlow<String?>(null)
    val error: StateFlow<String?> = _error.asStateFlow()

    private var codec: MediaCodec? = null
    private val pending = ConcurrentLinkedQueue<ByteArray>()
    // Indecsii buffer-elor de intrare libere, oferite de codec prin callback.
    private val freeInputs = ConcurrentLinkedQueue<Int>()

    @Synchronized
    fun start(surface: Surface, width: Int, height: Int) {
        if (codec != null) return
        _firstFrameRendered.value = false
        _error.value = null
        try {
            val format = MediaFormat.createVideoFormat(MediaFormat.MIMETYPE_VIDEO_AVC, width, height)
            val mc = MediaCodec.createDecoderByType(MediaFormat.MIMETYPE_VIDEO_AVC)
            mc.setCallback(callback)
            mc.configure(format, surface, null, 0)
            codec = mc
            mc.start()
        } catch (e: Exception) {
            _error.value = e.message ?: "Eroare initializare decodor"
            stop()
        }
    }

    fun submit(bytes: ByteArray) {
        pending.add(bytes)
        drain()
    }

    /** Trimite din coada in orice buffer de intrare liber. */
    private fun drain() {
        val mc = codec ?: return
        while (true) {
            val index = freeInputs.poll() ?: break
            val chunk = pending.poll()
            if (chunk == null) {
                // Nimic de trimis: punem indexul inapoi si iesim.
                freeInputs.add(index)
                break
            }
            try {
                val input = mc.getInputBuffer(index) ?: continue
                input.clear()
                input.put(chunk)
                mc.queueInputBuffer(index, 0, chunk.size, 0, 0)
            } catch (e: Exception) {
                _error.value = e.message ?: "Eroare trimitere cadru"
                break
            }
        }
    }

    @Synchronized
    fun stop() {
        val mc = codec ?: return
        codec = null
        runCatching { mc.stop() }
        runCatching { mc.release() }
        pending.clear()
        freeInputs.clear()
    }

    private val callback = object : MediaCodec.Callback() {
        override fun onInputBufferAvailable(mc: MediaCodec, index: Int) {
            freeInputs.add(index)
            drain()
        }

        override fun onOutputBufferAvailable(
            mc: MediaCodec,
            index: Int,
            info: MediaCodec.BufferInfo
        ) {
            // render=true elibereaza cadrul direct pe Surface.
            runCatching { mc.releaseOutputBuffer(index, true) }
            if (!_firstFrameRendered.value) _firstFrameRendered.value = true
        }

        override fun onError(mc: MediaCodec, e: MediaCodec.CodecException) {
            _error.value = e.message ?: "Eroare decodor"
        }

        override fun onOutputFormatChanged(mc: MediaCodec, format: MediaFormat) {
            // Nimic: randam pe Surface, nu ne intereseaza formatul de iesire.
        }
    }
}
```

```kotlin
package com.portalinstaller.viewmodel

import android.app.Application
import android.content.ContentValues
import android.content.Context
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.os.Build
import android.os.Environment
import android.provider.MediaStore
import android.view.Surface
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.portalinstaller.R
import com.portalinstaller.data.AppContainer
import com.portalinstaller.data.tools.H264Decoder
import com.portalinstaller.data.tools.InputInjector
import com.portalinstaller.data.tools.MirrorUnsupportedException
import com.portalinstaller.data.tools.PortalKey
import com.portalinstaller.data.tools.ScreenCaptureManager
import com.portalinstaller.data.tools.ScreenMirrorManager
import dadb.Dadb
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

sealed class MirrorState {
    data object Idle : MirrorState()
    data object Starting : MirrorState()
    data object Streaming : MirrorState()
    data class Unsupported(val message: String) : MirrorState()
    data class Error(val message: String) : MirrorState()
    data object NotConnected : MirrorState()
}

sealed class MirrorSaveState {
    data object Idle : MirrorSaveState()
    data object Saved : MirrorSaveState()
    data class Failed(val message: String) : MirrorSaveState()
}

/**
 * Orchestreaza oglindirea ecranului Portal: rezolutie -> decodor -> stream H.264,
 * randat pe Surface-ul din UI. Evenimentele de control (tap/swipe/taste/text)
 * sunt trimise ca apeluri `shell()` scurte. Pastreaza si salvarea unui cadru ca
 * PNG (mostenit din vechiul CaptureViewModel) folosind ScreenCaptureManager.
 */
class MirrorViewModel(app: Application) : AndroidViewModel(app) {

    private val holder = AppContainer.from(app).connectionHolder

    private val _state = MutableStateFlow<MirrorState>(MirrorState.Idle)
    val state: StateFlow<MirrorState> = _state.asStateFlow()

    private val _saveState = MutableStateFlow<MirrorSaveState>(MirrorSaveState.Idle)
    val saveState: StateFlow<MirrorSaveState> = _saveState.asStateFlow()

    private val _notConnected = MutableStateFlow(false)
    val notConnected: StateFlow<Boolean> = _notConnected.asStateFlow()

    private val _batteryWarning = MutableStateFlow(false)
    val batteryWarning: StateFlow<Boolean> = _batteryWarning.asStateFlow()

    private val decoder = H264Decoder()
    val firstFrameRendered: StateFlow<Boolean> = decoder.firstFrameRendered

    private var mirrorJob: Job? = null
    private var deviceWidth = 0
    private var deviceHeight = 0

    /** Chemat cand SurfaceView-ul are o suprafata gata: pornim oglindirea. */
    fun onSurfaceAvailable(surface: Surface) {
        if (mirrorJob != null) return
        val d = holder.dadb.value
        if (d == null) {
            _notConnected.value = true
            _state.value = MirrorState.NotConnected
            return
        }
        _notConnected.value = false
        _state.value = MirrorState.Starting
        var elapsedMs = 0L
        mirrorJob = viewModelScope.launch {
            try {
                val size = ScreenMirrorManager.deviceSize(d)
                deviceWidth = size.width
                deviceHeight = size.height
                decoder.start(surface, size.width, size.height)
                _state.value = MirrorState.Streaming
                launch { observeDecoderError() }
                launch { observeBattery() }
                ScreenMirrorManager.stream(d, size).collect { chunk ->
                    decoder.submit(chunk)
                }
            } catch (e: MirrorUnsupportedException) {
                _state.value = MirrorState.Unsupported(
                    e.message ?: getApplication<Application>().getString(R.string.mirror_unsupported)
                )
                decoder.stop()
            } catch (e: Exception) {
                _state.value = MirrorState.Error(
                    e.message ?: getApplication<Application>().getString(R.string.error_unknown)
                )
                decoder.stop()
            }
        }
    }

    private suspend fun observeDecoderError() {
        decoder.error.collect { message ->
            if (message != null) {
                _state.value = MirrorState.Error(message)
                stopMirror()
            }
        }
    }

    private suspend fun observeBattery() {
        // Avertisment de baterie dupa 5 minute de oglindire continua.
        delay(BATTERY_WARN_AFTER_MS)
        _batteryWarning.value = true
    }

    fun onSurfaceDestroyed() {
        stopMirror()
    }

    fun tap(viewX: Float, viewY: Float, viewW: Int, viewH: Int) {
        if (!firstFrameRendered.value) return
        val (x, y) = InputInjector.mapPoint(viewX, viewY, viewW, viewH, deviceWidth, deviceHeight)
        fireAndForget(InputInjector.tapCommand(x, y))
    }

    fun swipe(x1: Float, y1: Float, x2: Float, y2: Float, viewW: Int, viewH: Int, durationMs: Int) {
        if (!firstFrameRendered.value) return
        val (dx1, dy1) = InputInjector.mapPoint(x1, y1, viewW, viewH, deviceWidth, deviceHeight)
        val (dx2, dy2) = InputInjector.mapPoint(x2, y2, viewW, viewH, deviceWidth, deviceHeight)
        fireAndForget(InputInjector.swipeCommand(dx1, dy1, dx2, dy2, durationMs))
    }

    fun longPress(viewX: Float, viewY: Float, viewW: Int, viewH: Int) {
        if (!firstFrameRendered.value) return
        val (x, y) = InputInjector.mapPoint(viewX, viewY, viewW, viewH, deviceWidth, deviceHeight)
        fireAndForget(InputInjector.longPressCommand(x, y))
    }

    fun sendKey(key: PortalKey) {
        if (!firstFrameRendered.value) return
        fireAndForget(InputInjector.keyeventCommand(key))
    }

    fun sendText(raw: String) {
        if (raw.isEmpty() || !firstFrameRendered.value) return
        fireAndForget(InputInjector.textCommand(raw))
    }

    /** Trimite o comanda scurta `shell()` fara sa astepte rezultatul. */
    private fun fireAndForget(command: String) {
        val d = holder.dadb.value ?: return
        viewModelScope.launch(Dispatchers.IO) {
            runCatching { d.shell(command) }
        }
    }

    fun saveCurrentFrame(context: Context) {
        val d = holder.dadb.value
        if (d == null) {
            _notConnected.value = true
            return
        }
        viewModelScope.launch {
            val result = withContext(Dispatchers.IO) {
                runCatching {
                    val file = ScreenCaptureManager.capture(d, context.cacheDir)
                    val bitmap = BitmapFactory.decodeFile(file.absolutePath)
                        ?: throw IllegalStateException("Nu pot citi captura")
                    writeToMediaStore(context, bitmap)
                }
            }
            _saveState.value = result.fold(
                onSuccess = { MirrorSaveState.Saved },
                onFailure = { e ->
                    MirrorSaveState.Failed(e.message ?: context.getString(R.string.error_unknown))
                }
            )
        }
    }

    private fun writeToMediaStore(context: Context, bitmap: Bitmap) {
        val fileName = "portal_${System.currentTimeMillis()}.png"
        val values = ContentValues().apply {
            put(MediaStore.Images.Media.DISPLAY_NAME, fileName)
            put(MediaStore.Images.Media.MIME_TYPE, "image/png")
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                put(
                    MediaStore.Images.Media.RELATIVE_PATH,
                    "${Environment.DIRECTORY_PICTURES}/PortalToolkit"
                )
            }
        }
        val resolver = context.contentResolver
        val uri = resolver.insert(MediaStore.Images.Media.EXTERNAL_CONTENT_URI, values)
            ?: throw IllegalStateException("MediaStore insert failed")
        resolver.openOutputStream(uri)?.use { out ->
            bitmap.compress(Bitmap.CompressFormat.PNG, 100, out)
        } ?: throw IllegalStateException("MediaStore output stream unavailable")
    }

    fun clearSaveState() {
        _saveState.value = MirrorSaveState.Idle
    }

    fun clearNotConnected() {
        _notConnected.value = false
    }

    fun onDisconnected() {
        stopMirror()
        _notConnected.value = false
        _saveState.value = MirrorSaveState.Idle
    }

    override fun onCleared() {
        super.onCleared()
        stopMirror()
    }

    private fun stopMirror() {
        mirrorJob?.cancel()
        mirrorJob = null
        decoder.stop()
        _batteryWarning.value = false
        if (_state.value is MirrorState.Streaming || _state.value is MirrorState.Starting) {
            _state.value = MirrorState.Idle
        }
    }

    companion object {
        const val BATTERY_WARN_AFTER_MS = 5 * 60 * 1_000L
    }
}
```

```kotlin
/**
 * Ecranul de oglindire: gazduieste un SurfaceView (via AndroidView), preda
 * Surface-ul catre VM, detecteaza tap/drag/long-press si le trimite cu
 * dimensiunea masurata a view-ului. Arata un spinner pana la primul cadru; pe
 * Unsupported arata mesaj + buton de revenire la captura statica.
 */
@Composable
internal fun MirrorScreen(
    context: Context,
    viewModel: MirrorViewModel
) {
    val state by viewModel.state.collectAsStateWithLifecycle()
    val firstFrame by viewModel.firstFrameRendered.collectAsStateWithLifecycle()
    val batteryWarning by viewModel.batteryWarning.collectAsStateWithLifecycle()

    var viewSize by remember { mutableStateOf(IntSize.Zero) }
    var showKeyboard by remember { mutableStateOf(false) }
    var textToSend by remember { mutableStateOf("") }

    Column(modifier = Modifier.fillMaxSize()) {
        if (batteryWarning) {
            Text(
                text = stringResource(R.string.mirror_battery_note),
                modifier = Modifier.padding(8.dp)
            )
        }

        Box(
            modifier = Modifier.weight(1f).fillMaxWidth(),
            contentAlignment = Alignment.Center
        ) {
            when (val s = state) {
                is MirrorState.Unsupported -> {
                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
                        Text(s.message)
                        TextButton(onClick = { viewModel.saveCurrentFrame(context) }) {
                            Text(stringResource(R.string.mirror_fallback_peek))
                        }
                    }
                }
                is MirrorState.Error -> Text(s.message)
                is MirrorState.NotConnected ->
                    Text(stringResource(R.string.mirror_not_connected))
                else -> {
                    MirrorSurface(
                        onViewSize = { viewSize = it },
                        onSurface = { holder -> viewModel.onSurfaceAvailable(holder.surface) },
                        onSurfaceDestroyed = { viewModel.onSurfaceDestroyed() },
                        onTap = { o -> viewModel.tap(o.x, o.y, viewSize.width, viewSize.height) },
                        onDrag = { start, end ->
                            viewModel.swipe(
                                start.x, start.y, end.x, end.y,
                                viewSize.width, viewSize.height, 200
                            )
                        },
                        onLongPress = { o ->
                            viewModel.longPress(o.x, o.y, viewSize.width, viewSize.height)
                        }
                    )
                    if (!firstFrame) CircularProgressIndicator()
                }
            }
        }

        ControlBar(
            onKey = { key -> viewModel.sendKey(key) },
            onOpenKeyboard = { showKeyboard = true },
            onSaveFrame = { viewModel.saveCurrentFrame(context) }
        )
    }

    if (showKeyboard) {
        ModalBottomSheet(onDismissRequest = { showKeyboard = false }) {
            Column(modifier = Modifier.fillMaxWidth().padding(16.dp)) {
                OutlinedTextField(
                    value = textToSend,
                    onValueChange = { textToSend = it },
                    label = { Text(stringResource(R.string.mirror_text_hint)) },
                    modifier = Modifier.fillMaxWidth()
                )
                Row(
                    modifier = Modifier.fillMaxWidth().padding(top = 8.dp),
                    horizontalArrangement = Arrangement.End
                ) {
                    TextButton(onClick = {
                        viewModel.sendText(textToSend)
                        textToSend = ""
                        showKeyboard = false
                    }) {
                        Text(stringResource(R.string.mirror_text_send))
                    }
                }
            }
        }
    }
}

/**
 * SurfaceView-ul propriu-zis plus detectia gesturilor. Masuram dimensiunea
 * view-ului pentru maparea coordonatelor din VM.
 */
@Composable
private fun MirrorSurface(
    onViewSize: (IntSize) -> Unit,
    onSurface: (SurfaceHolder) -> Unit,
    onSurfaceDestroyed: () -> Unit,
    onTap: (Offset) -> Unit,
    onDrag: (Offset, Offset) -> Unit,
    onLongPress: (Offset) -> Unit
) {
    var dragStart by remember { mutableStateOf(Offset.Zero) }
    AndroidView(
        modifier = Modifier
            .fillMaxSize()
            .pointerInput(Unit) {
                onViewSize(IntSize(size.width, size.height))
                detectTapGestures(
                    onTap = { onTap(it) },
                    onLongPress = { onLongPress(it) }
                )
            }
            .pointerInput(Unit) {
                detectDragGestures(
                    onDragStart = { dragStart = it },
                    onDragEnd = { /* end pozitia curenta este trimisa in onDrag */ }
                ) { change, _ ->
                    onDrag(dragStart, change.position)
                }
            },
        factory = { ctx ->
            SurfaceView(ctx).apply {
                holder.addCallback(object : SurfaceHolder.Callback {
                    override fun surfaceCreated(holder: SurfaceHolder) = onSurface(holder)
                    override fun surfaceChanged(
                        holder: SurfaceHolder, format: Int, width: Int, height: Int
                    ) {
                        onViewSize(IntSize(width, height))
                    }
                    override fun surfaceDestroyed(holder: SurfaceHolder) = onSurfaceDestroyed()
                })
            }
        }
    )
}
```

```kotlin
package com.portalinstaller.data.screensaver

/** Numarul recomandat de poze unice pentru un slideshow variat pe Portal. */
const val RECOMMENDED_MIN = 20

/** Nivelul indiciului afisat pe tab-ul de screensaver. */
sealed class HintLevel {
    /** Fara indiciu (0 poze — starea goala acopera deja cazul). */
    data object None : HintLevel()

    /** Prea putine poze: sugereaza adaugarea a [needed] poze in plus. */
    data class AddMore(val needed: Int) : HintLevel()

    /** Suficiente poze: fara mesaj de tip „adauga mai multe”. */
    data object Ok : HintLevel()
}

/**
 * Firmware-ul Portal repeta seturile mici de poze; singura parghie reala este
 * un set mai mare de poze unice. Aceasta functie pura decide ce indiciu aratam
 * in functie de numarul de poze existente.
 */
fun screensaverHint(count: Int, recommendedMin: Int = RECOMMENDED_MIN): HintLevel = when {
    count <= 0 -> HintLevel.None
    count < recommendedMin -> HintLevel.AddMore(recommendedMin - count)
    else -> HintLevel.Ok
}
```
